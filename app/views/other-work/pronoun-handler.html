JavaScript — Person & Pronouns 
<script>
  const person = JSON.parse(sessionStorage.getItem('person') || '{}');

  const pronounForms = {
    he:   { subject: 'he',   object: 'him',  possessive: 'his' },
    she:  { subject: 'she',  object: 'her',  possessive: 'her' },
    they: { subject: 'they', object: 'them', possessive: 'their' },
    you:  { subject: 'you',  object: 'you',  possessive: 'your' }
  };

  const isFirstPerson = person.pronoun === 'you';
  const pronouns = pronounForms[person.pronoun] || pronounForms.you;

  const personWords = {
    name: isFirstPerson ? 'your' : person.fullName || '',
    nameEnding: isFirstPerson ? '' : "'s",

    subject: pronouns.subject,
    object: pronouns.object,
    possessive: pronouns.possessive,
    myPossessive: isFirstPerson ? 'my' : 'their',

    are: isFirstPerson ? 'are' : 'is',

    contraction: isFirstPerson ? "I'm" : "they're",
    contractionExpanded: isFirstPerson ? 'I am' : 'they are',

    Name: isFirstPerson ? 'Your' : person.fullName || '',
    NameEnding: isFirstPerson ? '' : "'s",

    Subject: capitalize(pronouns.subject),
    Object: capitalize(pronouns.object),
    Possessive: capitalize(pronouns.possessive),
    MyPossessive: isFirstPerson ? 'My' : 'Their',

    Are: isFirstPerson ? 'Are' : 'Is',

    Contraction: isFirstPerson ? "I'm" : "They're",
    ContractionExpanded: isFirstPerson ? 'I am' : 'They are'
  };

  function capitalize(word) {
    return word.charAt(0).toUpperCase() + word.slice(1);
  }

  document.querySelectorAll('[data-person-word]').forEach(function (element) {
    const word = element.dataset.personWord;
    element.textContent = personWords[word] || '';
  });
</script>
